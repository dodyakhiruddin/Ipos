Arsitektur Offline-First + Supabase Cloud: Semua transaksi & data user otomatis tersimpan di IndexedDB Browser (cepat & jalan tanpa internet). Jika URL & Anon Key Supabase diisi, data akan otomatis disinkronkan ke PostgreSQL Cloud saat online. 
-- ============================================================================
-- SKEMA SQL LENGKAP SUPABASE - DONET CELL [iPos 5.0 Web Edition]
-- Salin seluruh kode SQL ini dan jalankan di Menu "SQL Editor" Supabase Anda
-- ============================================================================

-- 1. Tabel Utama Penyimpanan State Multi-Database (Offline-First Sync)
CREATE TABLE IF NOT EXISTS public.donet_cell_state (
    id TEXT PRIMARY KEY,
    db_name TEXT NOT NULL DEFAULT 'GROSIRDONET',
    payload JSONB NOT NULL,
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- 2. Tabel Master Item & Stok Barang (Untuk Query & Laporan Terpisah)
CREATE TABLE IF NOT EXISTS public.donet_items (
    id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
    db_name TEXT NOT NULL,
    code TEXT NOT NULL,
    name TEXT NOT NULL,
    category TEXT,
    unit TEXT DEFAULT 'PCS',
    stock NUMERIC DEFAULT 0,
    cost NUMERIC DEFAULT 0,
    price NUMERIC DEFAULT 0,
    updated_at TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(db_name, code)
);

-- 3. Tabel Transaksi Penjualan POS, PPOB & BRILink (Mendukung Alur Sumber Dana & Tujuan Bayar)
CREATE TABLE IF NOT EXISTS public.donet_transactions (
    id TEXT PRIMARY KEY,
    db_name TEXT NOT NULL,
    trx_date TIMESTAMPTZ NOT NULL,
    module TEXT NOT NULL, -- 'POS', 'PPOB', 'BRILINK'
    sub_category TEXT,    -- 'pulsa', 'transfer', 'tarik', 'ewallet', 'pln'
    description TEXT,
    customer TEXT,
    provider TEXT,
    target_no TEXT,
    nominal NUMERIC DEFAULT 0,
    modal NUMERIC DEFAULT 0,
    admin_margin NUMERIC DEFAULT 0,
    total NUMERIC DEFAULT 0,
    profit NUMERIC DEFAULT 0,
    source_wallet TEXT,   -- 'SALDO_PULSA', 'SALDO_BRILINK', 'KAS_TOKO'
    dest_wallet TEXT,     -- 'KAS_TOKO', 'SALDO_BRILINK', 'SALDO_PULSA'
    payment_method TEXT,
    items JSONB,
    created_by TEXT,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Tambahkan kolom baru secara aman jika tabel donet_transactions sudah pernah dibuat sebelumnya:
ALTER TABLE public.donet_transactions ADD COLUMN IF NOT EXISTS sub_category TEXT;
ALTER TABLE public.donet_transactions ADD COLUMN IF NOT EXISTS nominal NUMERIC DEFAULT 0;
ALTER TABLE public.donet_transactions ADD COLUMN IF NOT EXISTS admin_margin NUMERIC DEFAULT 0;
ALTER TABLE public.donet_transactions ADD COLUMN IF NOT EXISTS source_wallet TEXT;
ALTER TABLE public.donet_transactions ADD COLUMN IF NOT EXISTS dest_wallet TEXT;

-- 4. Tabel Faktur Pembelian & Kulakan Stok
CREATE TABLE IF NOT EXISTS public.donet_purchases (
    id TEXT PRIMARY KEY,
    db_name TEXT NOT NULL,
    purchase_date TIMESTAMPTZ NOT NULL,
    dept TEXT DEFAULT 'UTM',
    supplier TEXT,
    source_wallet TEXT,
    sub_total NUMERIC DEFAULT 0,
    discount NUMERIC DEFAULT 0,
    tax NUMERIC DEFAULT 0,
    other_cost NUMERIC DEFAULT 0,
    total NUMERIC DEFAULT 0,
    paid NUMERIC DEFAULT 0,
    credit NUMERIC DEFAULT 0,
    items JSONB,
    created_by TEXT,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 5. Tabel Saldo Dompet & Akuntansi (Kas Toko, BRILink, Server Pulsa)
CREATE TABLE IF NOT EXISTS public.donet_wallets (
    db_name TEXT PRIMARY KEY,
    kas_toko NUMERIC DEFAULT 0,
    saldo_brilink NUMERIC DEFAULT 0,
    saldo_pulsa NUMERIC DEFAULT 0,
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- 6. Aktifkan Row Level Security (RLS) & Kebijakan Akses API
ALTER TABLE public.donet_cell_state ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.donet_items ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.donet_transactions ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.donet_purchases ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.donet_wallets ENABLE ROW LEVEL SECURITY;

DROP POLICY IF EXISTS "Allow anon all state" ON public.donet_cell_state;
CREATE POLICY "Allow anon all state" ON public.donet_cell_state FOR ALL USING (true) WITH CHECK (true);

DROP POLICY IF EXISTS "Allow anon all items" ON public.donet_items;
CREATE POLICY "Allow anon all items" ON public.donet_items FOR ALL USING (true) WITH CHECK (true);

DROP POLICY IF EXISTS "Allow anon all transactions" ON public.donet_transactions;
CREATE POLICY "Allow anon all transactions" ON public.donet_transactions FOR ALL USING (true) WITH CHECK (true);

DROP POLICY IF EXISTS "Allow anon all purchases" ON public.donet_purchases;
CREATE POLICY "Allow anon all purchases" ON public.donet_purchases FOR ALL USING (true) WITH CHECK (true);

DROP POLICY IF EXISTS "Allow anon all wallets" ON public.donet_wallets;
CREATE POLICY "Allow anon all wallets" ON public.donet_wallets FOR ALL USING (true) WITH CHECK (true);
